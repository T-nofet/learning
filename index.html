// קובץ Code.gs המלא והמאוחד - יצירת טוקנים אוטומטית וסנכרון לגיטהאב const CONFIG = { get SCHEDULE_SS_ID() { return PropertiesService.getScriptProperties().getProperty('SCHEDULE_SS_ID'); }, get REPORTS_SS_ID() { return PropertiesService.getScriptProperties().getProperty('REPORTS_SS_ID'); }, get GITHUB_TOKEN() { return PropertiesService.getScriptProperties().getProperty('GITHUB_TOKEN'); }, get GITHUB_REPO() { return PropertiesService.getScriptProperties().getProperty('GITHUB_REPO'); }, // פורמט: username/repo DAYS_OF_WEEK: ["ראשון", "שני", "שלישי", "רביעי", "חמישי"], STUDENTS_SHEET_NAME: "רשימת נערי יחידה" }; // פונקציית הטריגר המרכזית - מריצים אותה כדי לעדכן את גיטהאב function exportAndSyncToGitHub() { try { // 1. קודם כל בודקים ומייצרים טוקנים חסרים בגיליון השעות generateMissingTokens(); // 2. אוספים את כל הנתונים המעודכנים (כולל הטוקנים החדשים שנוצרו) const payloadData = getCombinedRawData(); const githubUrl = `https://api.github.com/repos/${CONFIG.GITHUB_REPO}/contents/data.json`; // 3. בדיקה אם הקובץ כבר קיים בגיטהאב כדי לקבל את ה-SHA שלו (חובה ב-API של GitHub) let sha = null; const checkResponse = UrlFetchApp.fetch(githubUrl, { method: "GET", headers: { "Authorization": "token " + CONFIG.GITHUB_TOKEN }, muteHttpExceptions: true }); if (checkResponse.getResponseCode() === 200) { sha = JSON.parse(checkResponse.getContentText()).sha; } // 4. הכנת הבקשה לעדכון הקובץ const commitBody = { message: "🔄 עדכון אוטומטי של נתוני מרכז למידה טוקנים ומערכות", content: Utilities.base64Encode(JSON.stringify(payloadData, null, 2), Utilities.Charset.UTF_8) }; if (sha) commitBody.sha = sha; const options = { method: "PUT", headers: { "Authorization": "token " + CONFIG.GITHUB_TOKEN, "Content-Type": "application/json" }, payload: JSON.stringify(commitBody), muteHttpExceptions: true }; const uploadResponse = UrlFetchApp.fetch(githubUrl, options); Logger.log("GitHub Response: " + uploadResponse.getContentText()); } catch (err) { Logger.log("Error Syncing to GitHub: " + err.message); } } // בדיקת תלמידים חדשים מה-IMPORTRANGE ויצירת טוקנים קבועים עבורם בשורה המתאימה function generateMissingTokens() { try { const scheduleSS = SpreadsheetApp.openById(CONFIG.SCHEDULE_SS_ID); const sheet = scheduleSS.getSheetByName(CONFIG.STUDENTS_SHEET_NAME); if (!sheet) { Logger.log("שגיאה ביצירת טוקנים: לא נמצא גיליון בשם " + CONFIG.STUDENTS_SHEET_NAME); return; } var lastRow = sheet.getLastRow(); if (lastRow < 2) return; // עמודה A = שם פרטי (1), עמודה H = טוקן (8) var nameColumn = 1; var tokenColumn = 8; var names = sheet.getRange(2, nameColumn, lastRow - 1, 1).getValues(); var tokensRange = sheet.getRange(2, tokenColumn, lastRow - 1, 1); var tokens = tokensRange.getValues(); var updated = false; for (var i = 0; i < names.length; i++) { var studentName = names[i][0].toString().trim(); var currentToken = tokens[i][0].toString().trim(); // אם נוצר שם חדש ב-IMPORTRANGE ועמודת הטוקן שלו בגיליון ריקה if (studentName !== "" && currentToken === "") { tokens[i][0] = makeRandomTokenString(6); updated = true; Logger.log(`מפתח חדש נוצר עבור: ${studentName}`); } } if (updated) { tokensRange.setValues(tokens); SpreadsheetApp.flush(); // דוחף את השינוי פיזית לתוך הגוגל שיטס מיד } } catch (err) { Logger.log("Error generating tokens: " + err.message); } } // פונקציית עזר להגרלת מחרוזת אקראית באנגלית (אותיות גדולות ומספרים) function makeRandomTokenString(length) { var result = ''; var characters = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789'; for (var i = 0; i < length; i++) { result += characters.charAt(Math.floor(Math.random() * characters.length)); } return result; } // איסוף ועיבוד הנתונים הגולמיים מהטבלאות לאחר עדכון הטוקנים function getCombinedRawData() { const scheduleSS = SpreadsheetApp.openById(CONFIG.SCHEDULE_SS_ID); const reportsSS = SpreadsheetApp.openById(CONFIG.REPORTS_SS_ID); // איסוף מערכת שעות const schedule = {}; CONFIG.DAYS_OF_WEEK.forEach(day => { const sheet = scheduleSS.getSheetByName(day); schedule[day] = sheet ? sheet.getDataRange().getDisplayValues() : []; }); // איסוף רשימת תלמידים (שם מול טוקן) const studentSheet = scheduleSS.getSheetByName(CONFIG.STUDENTS_SHEET_NAME); const studentData = studentSheet ? studentSheet.getDataRange().getDisplayValues() : []; const studentsMapping = {}; if (studentData.length > 1) { for (let i = 1; i < studentData.length; i++) { if (studentData[i][0] && studentData[i][7]) { studentsMapping[studentData[i][7].toString().trim()] = studentData[i][0].toString().trim(); } } } // איסוף וסינון דיווחים const reportSheet = reportsSS.getSheets()[0]; const reportData = reportSheet ? reportSheet.getDataRange().getDisplayValues() : []; const months = new Set(), teachers = new Set(); const reports = reportData.slice(1).filter(r => r[0] !== "").map(row => { const ts = row[0].toString(); let monthLabel = "כללי"; if (ts.includes('/')) { const parts = ts.split('/'); if (parts.length >= 3) { monthLabel = parts[1].padStart(2, '0') + '/' + parts[2].split(' ')[0]; months.add(monthLabel); } } if (row[1]) teachers.add(row[1].trim()); return { timestamp: ts, month: monthLabel, teacher: row[1] || "לא ידוע", studentName: row[2] || "", content: row.slice(3, 8).join(" | "), allText: row.join(" ").toLowerCase() }; }); return { schedule, reports, studentsMapping, availableMonths: Array.from(months).sort().reverse(), availableTeachers: Array.from(teachers).sort(), adminPw: PropertiesService.getScriptProperties().getProperty('ADMIN_PW'), studentPw: PropertiesService.getScriptProperties().getProperty('STUDENT_PW') }; }
